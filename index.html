$('create').onclick=async()=>{try{let sid=$('sid').value.trim().toUpperCase(),name=$('sname').value.trim(),pw=$('spass').value;if(!sid||!name||!pw)throw Error('तीनों बॉक्स भरें।');if(pw.length<8)throw Error('Password कम से कम 8 अक्षर का रखें।');let {data:{session}}=await sb.auth.getSession();if(!session)throw Error('Admin login दोबारा करें।');let {data,error}=await sb.functions.invoke('clever-processor',{body:{student_id:sid,full_name:name,password:pw},headers:{Authorization:'Bearer '+session.access_token}});if(error){let d=error.message;if(error.context){try{let b=await error.context.json();d+=' | '+(b.error||JSON.stringify(b))}catch(_){}}throw Error(d)}if(data?.error)throw Error(data.error);msg('Student account बन गया: '+sid,true);$('sid').value='';$('sname').value='';$('spass').value='';await refreshList()}catch(e){msg('Student create error: '+e.message)}};
